document.querySelectorAll(".case-toggle").forEach(button => { // Mobile Safari fix button.addEventListener("touchstart", () => { button.click(); }); button.addEventListener("click", () => { const card = button.closest(".case-card"); if (!card) return; const expanded = card.classList.toggle("expanded"); button.setAttribute("aria-expanded", expanded ? "true" : "false"); button.querySelector("span").textContent = expanded ? "−" : "+"; button.firstChild.textContent = expanded ? "Hide detail " : "View outcome "; }); });
